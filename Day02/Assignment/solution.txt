Department d

create view vw_billing
as
select ClaimId,BilledAmount ,[ReimbursedAmt],[Status] from [dbo].[Claim]

SELECT *
FROM Claim
ORDER BY ClaimId
OFFSET 0 ROWS FETCH NEXT 20 ROWS ONLY;

create procedure sp_RevenueLeakageReport
as
Begin
select Status , Count(*) as TotalClaims, Sum(BilledAmount) as TotalBilledAmt , 
Sum(ISNULL(ReimbursedAmt,0)) as TotalReimbursedAmt,
Sum(BilledAmount -ISNULL(ReimbursedAmt,0))as outstandingAmt , Rank() over(order by 
 Sum(BilledAmount -ISNULL(ReimbursedAmt,0)) desc) as lossRank
 from vw_billing
 group by Status
 order by lossRank
 End;

 EXEC sp_RevenueLeakageReport;

create or alter procedure sp_ExecutiveDashboard
as
begin
set nocount on;
select Count(*) as TotalActivePatients from Patient where IsActive=1;
select top 5 DepartmentId , count(*) as TotalEncounters from Encounter group by DepartmentId order by 
TotalEncounters desc;
select Avg(DATEDIFF(day,AdmitDate,DischargeDate)) as AverageLengthOfStay from Encounter;

with tab1 as (
select patientId ,AdmitDate, Lag(DischargeDate) over
(partition by patientId order by AdmitDate) as PreviousDischargeDate from Encounter where DateDiff(day,AdmitDate,GetDate())<=30
)

select count(*) as TotalReadmissionsLast30Days from tab1 where
PreviousDischargeDate   is not null and DATEDIFF(day ,PreviousDischargeDate,AdmitDate)<=15

end;

exec sp_ExecutiveDashboard

as
begin
